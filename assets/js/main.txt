// ---- Auto-dismiss Flash Alerts after 4 seconds ----
document.addEventListener('DOMContentLoaded', () => {
    document.querySelectorAll('.alert-auto').forEach(alert => {
        setTimeout(() => {
            alert.classList.remove('show');
            setTimeout(() => alert.remove(), 400);
        }, 4000);
    });
});


// ---- Email Validation Helper ----
function isValidEmail(email) {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}


// ---- Password Strength Indicator ----
const passwordInput = document.getElementById('password');
const strengthBar   = document.getElementById('password-strength');

if (passwordInput && strengthBar) {
    passwordInput.addEventListener('input', function () {
        const val = this.value;
        let strength = 0;

        if (val.length >= 8)          strength++;
        if (/[A-Z]/.test(val))        strength++;
        if (/[0-9]/.test(val))        strength++;
        if (/[^A-Za-z0-9]/.test(val)) strength++;

        const levels = ['danger', 'warning', 'info', 'success'];
        const labels = ['Weak', 'Fair', 'Good', 'Strong'];

        strengthBar.className = `progress-bar bg-${levels[strength - 1] || 'danger'}`;
        strengthBar.style.width = (strength * 25) + '%';
        strengthBar.textContent = labels[strength - 1] || '';
    });
}


// ---- Register Form Validation ----
const registerForm = document.getElementById('registerForm');
if (registerForm) {
    registerForm.addEventListener('submit', function (e) {
        const name     = document.getElementById('name')?.value.trim();
        const email    = document.getElementById('email')?.value.trim();
        const password = document.getElementById('password')?.value;
        const confirm  = document.getElementById('confirm_password')?.value;

        if (!name || name.length < 2) {
            e.preventDefault();
            showFormError('Please enter your full name (at least 2 characters).');
            return;
        }
        if (!isValidEmail(email)) {
            e.preventDefault();
            showFormError('Please enter a valid email address.');
            return;
        }
        if (password.length < 8) {
            e.preventDefault();
            showFormError('Password must be at least 8 characters long.');
            return;
        }
        if (password !== confirm) {
            e.preventDefault();
            showFormError('Passwords do not match. Please try again.');
            return;
        }
    });
}


// ---- Login Form Validation ----
const loginForm = document.getElementById('loginForm');
if (loginForm) {
    loginForm.addEventListener('submit', function (e) {
        const email    = document.getElementById('email')?.value.trim();
        const password = document.getElementById('password')?.value;

        if (!isValidEmail(email)) {
            e.preventDefault();
            showFormError('Please enter a valid email address.');
            return;
        }
        if (!password) {
            e.preventDefault();
            showFormError('Please enter your password.');
            return;
        }
    });
}


// ---- Show Form Error (inserts alert above the form) ----
function showFormError(message) {
    // Remove any existing error first
    document.querySelectorAll('.js-error-alert').forEach(a => a.remove());

    const alert = document.createElement('div');
    alert.className = 'alert alert-danger alert-dismissible fade show js-error-alert';
    alert.innerHTML = `
        <i class="bi bi-exclamation-triangle-fill me-2"></i>${message}
        <button type="button" class="btn-close" data-bs-dismiss="alert"></button>
    `;

    const form = document.querySelector('form');
    if (form) form.parentNode.insertBefore(alert, form);

    // Scroll to alert
    alert.scrollIntoView({ behavior: 'smooth', block: 'center' });
}


// ---- Booking Form: Calculate Total Cost ----
const durationInput   = document.getElementById('duration_days');
const groupInput      = document.getElementById('group_size');
const pricePerDay     = parseFloat(document.getElementById('price_per_day')?.value || 0);
const totalCostDisplay = document.getElementById('total_cost_display');

function updateTotal() {
    if (!durationInput || !groupInput || !totalCostDisplay) return;
    const days  = parseInt(durationInput.value) || 1;
    const group = parseInt(groupInput.value) || 1;
    const total = pricePerDay * days * group;
    totalCostDisplay.textContent = 'KES ' + total.toLocaleString();
}

if (durationInput) durationInput.addEventListener('input', updateTotal);
if (groupInput)    groupInput.addEventListener('input', updateTotal);
